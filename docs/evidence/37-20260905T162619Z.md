## The ad-hoc check, recorded so it is reproducible

The `NEVER-35.7` result above is only evidence if it can be re-run. The script is
therefore reproduced verbatim rather than described. It is deliberately **not
committed**: `duplicate-contract` is #280's rule, and shipping a second
implementation of a contract is the exact failure that rule exists to prevent.

```ts
import { readFileSync } from "node:fs";
import { join } from "node:path";
const root = "<repo root>";
const docs = ["verification","workflow-states","branch-and-pr-model"]
  .map(n => `references/${n}.md`);

const { SENTINEL_NAMESPACE_PREFIX } = await import(join(root,"packages/core/src/evidence/sentinel.ts"));
const { EXCLUSIVE_LABEL_PREFIXES }  = await import(join(root,"packages/core/src/gh/labels.ts"));
const { SUB_ISSUE_FEATURE_HEADER }  = await import(join(root,"packages/core/src/gh/sub-issues.ts"));
const { BACKOFF_BASE_MS, BACKOFF_CAP_MS } = await import(join(root,"packages/core/src/gh/resume.ts"));

const checks: Array<[string,string]> = [
  ["SENTINEL_NAMESPACE_PREFIX", SENTINEL_NAMESPACE_PREFIX],
  ["SUB_ISSUE_FEATURE_HEADER", SUB_ISSUE_FEATURE_HEADER],
  ["EXCLUSIVE_LABEL_PREFIXES(literal array)", JSON.stringify(EXCLUSIVE_LABEL_PREFIXES)],
  ["BACKOFF_BASE_MS", String(BACKOFF_BASE_MS)],
  ["BACKOFF_CAP_MS", String(BACKOFF_CAP_MS)],
];
// assert the denominator: a no-restatement check over an empty constant
// set is a rule that cannot fail.
if (checks.length < 4) throw new Error("denominator too small");

for (const d of docs) {
  const body = readFileSync(join(root,d),"utf8");
  for (const [name,val] of checks) {
    if (val && val.length > 2 && body.includes(val)) console.log(`VIOLATION ${d}: ${name}`);
  }
}
```
